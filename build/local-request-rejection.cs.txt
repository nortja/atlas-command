    // Reject before parsing or dispatching. Drain only a small framed body so an
    // unread POST does not turn the already-written 403 into a TCP reset.
    static void RejectUnmarkedRequest(TcpClient client, NetworkStream stream, Dictionary<string,string> headers)
    {
        Error(stream,403,"Local action header required");
        try { client.Client.Shutdown(SocketShutdown.Send); } catch (SocketException) { return; }
        int length; string value;
        if (!headers.TryGetValue("Content-Length",out value) || !Int32.TryParse(value,NumberStyles.None,CultureInfo.InvariantCulture,out length) || length <= 0 || length > 8192) return;
        byte[] discard = new byte[Math.Min(length,1024)];
        var deadline = Stopwatch.StartNew();
        try {
            while (length > 0 && deadline.ElapsedMilliseconds < 250) {
                stream.ReadTimeout = Math.Max(1,250-(int)deadline.ElapsedMilliseconds);
                int count=stream.Read(discard,0,Math.Min(length,discard.Length));
                if (count <= 0) break;
                length-=count;
            }
        } catch (IOException) { }
        finally { Array.Clear(discard,0,discard.Length); }
    }
