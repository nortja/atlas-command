    static string WorkspaceRevision(string text)
    {
        using (var hash = System.Security.Cryptography.SHA256.Create()) {
            return BitConverter.ToString(hash.ComputeHash(Utf8.GetBytes(text))).Replace("-", "").ToLowerInvariant();
        }
    }
    static string ReadWorkspaceText() { return File.Exists(StatePath) ? File.ReadAllText(StatePath,Utf8) : "null"; }
    static string RecoveryCurrentRevision()
    {
        if(!File.Exists(StatePath))return WorkspaceRevision("null");
        var info=new FileInfo(StatePath);
        if(info.Length>MaxJson||(info.Attributes&FileAttributes.ReparsePoint)!=0)throw new InvalidDataException("Workspace requires manual recovery.");
        try{return WorkspaceRevision(ReadWorkspaceText());}
        catch(DecoderFallbackException){using(var hash=SHA256.Create())return "bytes-"+BitConverter.ToString(hash.ComputeHash(File.ReadAllBytes(StatePath))).Replace("-","").ToLowerInvariant();}
    }
    static void WorkspaceHealth(NetworkStream stream)
    {
        lock(DataLock){
            string revision=RecoveryCurrentRevision(),status=File.Exists(StatePath)?"invalid":"missing";
            if(status!="missing")try{string text=ReadWorkspaceText();var parsed=Json().DeserializeObject(text) as Dictionary<string,object>;
                if(parsed!=null&&parsed.ContainsKey("schemaVersion")&&Convert.ToString(parsed["schemaVersion"],CultureInfo.InvariantCulture)!="1")status="unsupported";
                else {ValidateState(text);status="ready";}
            }catch{}
            SendJson(stream,Json().Serialize(new {status=status,revision=revision,bytes=File.Exists(StatePath)?new FileInfo(StatePath).Length:0}));
        }
    }
    static void PreserveRecoveryOriginal()
    {
        if(!File.Exists(StatePath))return;
        // A separate, never-overwritten original survives later ordinary saves.
        if(Directory.GetFiles(DataDir,"recovery-original-*.json").Length>=50)throw new InvalidDataException("Preserved recovery copies require review.");
        string path=Path.Combine(DataDir,"recovery-original-"+DateTime.UtcNow.ToString("yyyyMMddHHmmss",CultureInfo.InvariantCulture)+"-"+Guid.NewGuid().ToString("N")+".json");
        File.Copy(StatePath,path,false);
        using(var file=new FileStream(path,FileMode.Open,FileAccess.ReadWrite,FileShare.None)){file.Flush(true);}
    }
    static string RecordBackupPath(string id)
    {
        if(id=="previous") return StatePath+".bak";
        if(!Regex.IsMatch(id??"",@"\A(?:backup-[0-9]{4}-[0-9]{2}-[0-9]{2}|recovery-original-[0-9]{14}-[a-f0-9]{32})\.json\z"))throw new InvalidDataException("Unknown record backup.");
        return Path.Combine(DataDir,id);
    }
    static object RecoveryMediaSummary(string text)
    {
        int present=0,missing=0,unavailable=0;var seen=new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        var value=Json().DeserializeObject(text) as Dictionary<string,object>;
        foreach(object item in (object[])value["sites"]){var site=item as Dictionary<string,object>;object raw;
            if(!site.TryGetValue("media",out raw)||!(raw is object[]))continue;
            foreach(object entry in (object[])raw){var media=entry as Dictionary<string,object>;object key;
                if(media==null||!media.TryGetValue("key",out key)||!(key is string)){unavailable++;continue;}
                if(!seen.Add((string)key))continue;
                if(seen.Count>5000){unavailable++;continue;}
                try{var info=new FileInfo(MediaPath((string)key));if(!info.Exists)missing++;else if((info.Attributes&FileAttributes.ReparsePoint)!=0)unavailable++;else present++;}catch{unavailable++;}
            }
        }
        return new {present=present,missing=missing,unavailable=unavailable};
    }
    static string ReadRecordBackup(string id)
    {
        string path=RecordBackupPath(id);var info=new FileInfo(path);
        if(!info.Exists||info.Length>MaxJson||(info.Attributes&FileAttributes.ReparsePoint)!=0)throw new InvalidDataException("Record backup unavailable.");
        string text=File.ReadAllText(path,Utf8);ValidateState(text);return text;
    }
    static void HandleGuardedWorkspace(NetworkStream stream,string method,Dictionary<string,string> headers,byte[] body)
    {
        lock(DataLock) {
            if(File.Exists(StatePath)){var info=new FileInfo(StatePath);if(info.Length>MaxJson||(info.Attributes&FileAttributes.ReparsePoint)!=0)throw new InvalidDataException("Workspace requires manual recovery.");}
            string current=ReadWorkspaceText(),revision=WorkspaceRevision(current);
            if(method=="GET") { if(File.Exists(StatePath))ValidateState(current);SendJson(stream,"{\"revision\":\""+revision+"\",\"value\":"+current+"}");return; }
            if(method!="POST"){Error(stream,405,"Unsupported workspace action");return;}
            RequireType(headers,"application/json");var request=Json().DeserializeObject(Utf8.GetString(body)) as Dictionary<string,object>;
            if(request==null||!request.ContainsKey("revision")||!request.ContainsKey("value")){Error(stream,400,"Workspace revision required");return;}
            if(!String.Equals(Convert.ToString(request["revision"],CultureInfo.InvariantCulture),revision,StringComparison.Ordinal)){Error(stream,409,"Workspace changed; review before saving");return;}
            string text=Json().Serialize(request["value"]);ValidateState(text);SaveAtomic(text);
            SendJson(stream,"{\"revision\":\""+WorkspaceRevision(text)+"\"}");
        }
    }
    static void HandleRecordBackups(NetworkStream stream,string method,Dictionary<string,string> headers,byte[] body,string requestPath)
    {
        lock(DataLock) {
            if(method=="GET") {
                string query=new Uri(Origin+requestPath).Query;
                if(query.StartsWith("?review=",StringComparison.Ordinal)) {
                    string id=Uri.UnescapeDataString(query.Substring(8)),text=ReadRecordBackup(id);
                    SendJson(stream,Json().Serialize(new {id=id,revision=WorkspaceRevision(text),value=Json().DeserializeObject(text),media=RecoveryMediaSummary(text)}));return;
                }
                if(query.Length!=0){Error(stream,400,"Unknown backup query");return;}
                var entries=new List<object>();var ids=new List<string>();if(File.Exists(StatePath+".bak"))ids.Add("previous");
                foreach(string file in Directory.GetFiles(DataDir,"backup-*.json"))if(Regex.IsMatch(Path.GetFileName(file),@"\Abackup-[0-9]{4}-[0-9]{2}-[0-9]{2}\.json\z"))ids.Add(Path.GetFileName(file));
                foreach(string file in Directory.GetFiles(DataDir,"recovery-original-*.json"))if(Regex.IsMatch(Path.GetFileName(file),@"\Arecovery-original-[0-9]{14}-[a-f0-9]{32}\.json\z"))ids.Add(Path.GetFileName(file));
                ids.Sort(StringComparer.Ordinal);ids.Reverse();if(ids.Count>366)ids=ids.GetRange(0,366);
                foreach(string id in ids){var info=new FileInfo(RecordBackupPath(id));if((info.Attributes&FileAttributes.ReparsePoint)==0)entries.Add(new {id=id,bytes=info.Length,savedUtc=info.LastWriteTimeUtc.ToString("o",CultureInfo.InvariantCulture)});}
                SendJson(stream,Json().Serialize(entries));return;
            }
            if(method!="POST"){Error(stream,405,"Unsupported backup action");return;}
            RequireType(headers,"application/json");var request=Json().DeserializeObject(Utf8.GetString(body)) as Dictionary<string,object>;
            if(request==null||!request.ContainsKey("revision")||!request.ContainsKey("backupRevision")||!request.ContainsKey("id")){Error(stream,400,"Reviewed backup required");return;}
            string currentRevision=RecoveryCurrentRevision();
            if(!String.Equals(Convert.ToString(request["revision"],CultureInfo.InvariantCulture),currentRevision,StringComparison.Ordinal)){Error(stream,409,"Workspace changed; review again");return;}
            string restoredText=ReadRecordBackup(Convert.ToString(request["id"],CultureInfo.InvariantCulture));
            if(!String.Equals(Convert.ToString(request["backupRevision"],CultureInfo.InvariantCulture),WorkspaceRevision(restoredText),StringComparison.Ordinal)){Error(stream,409,"Backup changed; review again");return;}
            PreserveRecoveryOriginal();SaveAtomic(restoredText);SendJson(stream,"{\"revision\":\""+WorkspaceRevision(restoredText)+"\"}");
        }
    }
